 document.getElementById('labelPickerList').innerHTML=labels.length?labels.map(l=>`<label class="label-option"><input type="checkbox" value="${esc(l)}" ${pendingLabels.includes(l)?'checked':''}><span class="label-dot" style="background:${labelColor(l)}"></span><span>${esc(l)}</span></label>`).join(''):'<div style="padding:10px;color:#8b9096">No labels yet</div>';
 document.getElementById('labelOverlay').classList.add('show');
}
function applyLabels(){pendingLabels=[...document.querySelectorAll('#labelPickerList input[type=checkbox]:checked')].map(x=>x.value);renderEditorLabels(pendingLabels);if(currentId){const n=notes.find(x=>x.id===currentId);if(n){n.labels=[...pendingLabels];n.updated=Date.now();persist();renderLabels();renderNotes()}}hideLabelPicker();document.getElementById('saveStatus').textContent='Unsaved'}
function createAndAssignLabel(){const input=document.getElementById('newLabelInput'),name=input.value.trim();if(!name)return;if(!labels.includes(name))labels.push(name);pendingLabels=Array.from(new Set([...pendingLabels,name]));input.value='';persist();openLabelPicker();renderEditorLabels(pendingLabels);renderLabels();toast('Label added')}
function hideLabelPicker(){document.getElementById('labelOverlay').classList.remove('show')}
function closeLabelPicker(e){if(e.target===e.currentTarget)hideLabelPicker()}
function openLabelManager(){renderLabelManager();document.getElementById('labelManagerOverlay').classList.add('show')}
function renderLabelManager(){const box=document.getElementById('labelManagerList');box.innerHTML=labels.length?labels.map((l,i)=>`<div class="label-row"><span class="label-dot" style="background:${labelColor(l)}"></span><span class="label-name">${esc(l)}</span><button class="tiny-btn" title="Delete label" onclick="deleteLabel(${i})">${ICONS.trash}</button></div>`).join(''):'<div style="padding:10px;color:#8b9096">No labels yet</div>'}
function createManagedLabel(){const input=document.getElementById('managerLabelInput'),name=input.value.trim();if(!name)return;if(labels.includes(name)){toast('Label already exists');return}labels.push(name);input.value='';persist();renderLabels();renderLabelManager();toast('Label created')}
function deleteLabel(index){const name=labels[index];if(!name)return;if(!confirm(`Delete label “${name}”? Notes will keep their text but lose this label.`))return;labels.splice(index,1);notes.forEach(n=>{n.labels=n.labels.filter(l=>l!==name);n.updated=Date.now()});if(filter==='label:'+name)setFilter('notes',document.querySelector('[data-filter="notes"]'));persist();renderLabels();renderLabelManager();renderNotes();toast('Label deleted')}
function hideLabelManager(){document.getElementById('labelManagerOverlay').classList.remove('show')}
function closeLabelManager(e){if(e.target===e.currentTarget)hideLabelManager()}

function openReminder(){const n=currentId?notes.find(x=>x.id===currentId):null;const d=n?.reminder?new Date(n.reminder):new Date(Date.now()+3600000);document.getElementById('remDate').value=toLocalDateInput(d);document.getElementById('remTime').value=toLocalTimeInput(d);document.getElementById('reminderOverlay').classList.add('show')}
function hideReminder(){document.getElementById('reminderOverlay').classList.remove('show')}
function closeReminder(e){if(e.target===e.currentTarget)hideReminder()}
function saveReminder(){if(!currentId){toast('Save the note first, then add a reminder');hideReminder();return}const d=document.getElementById('remDate').value,t=document.getElementById('remTime').value;if(!d||!t){toast('Choose date and time');return}const n=notes.find(x=>x.id===currentId);n.reminder=new Date(d+'T'+t).toISOString();n.updated=Date.now();persist();updateReminderUI(n.reminder);renderNotes();hideReminder();toast('Reminder set')}
function clearReminder(){if(!currentId)return;const n=notes.find(x=>x.id===currentId);if(!n)return;n.reminder='';n.updated=Date.now();persist();updateReminderUI('');renderNotes();toast('Reminder removed')}
function updateReminderUI(value){const box=document.getElementById('reminderText');if(!value){box.style.display='none';document.getElementById('reminderValue').textContent='';return}box.style.display='flex';document.getElementById('reminderValue').textContent=new Date(value).toLocaleString()}
function toLocalDateInput(d){return`${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`}
function toLocalTimeInput(d){return`${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}`}

function exportNotes(){const data={version:2,exportedAt:new Date().toISOString(),labels,notes};const blob=new Blob([JSON.stringify(data,null,2)],{type:'application/json'});const a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download='notes-backup-'+new Date().toISOString().slice(0,10)+'.json';a.click();setTimeout(()=>URL.revokeObjectURL(a.href),1000);toast('Backup exported')}
function importNotes(e){const file=e.target.files?.[0];if(!file)return;const reader=new FileReader();reader.onload=()=>{try{const data=JSON.parse(reader.result);if(!Array.isArray(data.notes))throw new Error('Invalid backup');const incoming=data.notes.map(migrateNote);const choice=confirm('OK = replace existing notes. Cancel = merge backup into current notes.');if(choice){notes=incoming}else{const ids=new Set(notes.map(n=>n.id));incoming.forEach(n=>{if(ids.has(n.id))n.id=Date.now()+Math.floor(Math.random()*100000);n.updated=Date.now();notes.push(n)})}if(Array.isArray(data.labels))labels=Array.from(new Set([...labels,...data.labels.map(String).filter(Boolean)]));persist();renderLabels();renderNotes();toast('Backup imported')}catch(err){toast('Invalid backup file')}};reader.readAsText(file);e.target.value='' }

function toggleSidebar(force){const s=document.getElementById('sidebar'),b=document.getElementById('backdrop'),open=typeof force==='boolean'?force:!s.classList.contains('open');s.classList.toggle('open',open);b.classList.toggle('show',open)}
function toggleDark(){const dark=!document.body.classList.contains('dark');document.body.classList.toggle('dark',dark);localStorage.setItem('notes-dark',dark?'1':'0')}
