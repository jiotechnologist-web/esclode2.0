    19→function formatTime(time: number): string {
    20→  if (!isFinite(time) || time < 0) return "00:00";
    21→  let seconds = Math.floor(time % 60);
    22→  let minutes = Math.floor(time / 60) % 60;
    23→  let hours = Math.floor(time / 3600);
    24→  const pad = (n: number) => (n < 10 ? `0${n}` : `${n}`);
    25→  if (hours === 0) return `${pad(minutes)}:${pad(seconds)}`;
    26→  return `${pad(hours)}:${pad(minutes)}:${pad(seconds)}`;
    27→}

   385→/**
   386→ * ReelsFeed — vertical scroll-snap feed (TikTok/Instagram style).
   387→ * Each video occupies the full viewport. IntersectionObserver autoplays the
   388→ * active video and pauses the others. Next video is preloaded.
   389→ */
   390→function ReelsFeed({ items, startIndex, onClose }: { items: ApiMediaItem[]; startIndex: number; onClose: () => void }) {
   391→  const containerRef = useRef<HTMLDivElement>(null);
   392→  const itemRefs = useRef<(HTMLDivElement | null)[]>([]);
   393→  const [activeIndex, setActiveIndex] = useState(startIndex);
   394→  const [showChrome, setShowChrome] = useState(true);
   395→  const hideChromeTimer = useRef<any>(null);
   396→
   397→  // Initial scroll
   398→  useEffect(() => {
   399→    const el = itemRefs.current[startIndex];
   400→    if (el) el.scrollIntoView({ behavior: "auto", block: "start" });
   401→  }, [startIndex]);
   402→
   403→  // IntersectionObserver — track which item is active
   404→  useEffect(() => {
   405→    const opts: IntersectionObserverInit = { root: containerRef.current, threshold: [0.6] };
   406→    const obs = new IntersectionObserver((entries) => {
   407→      for (const e of entries) {
   408→        const idx = Number((e.target as HTMLElement).dataset.index);
   409→        if (e.isIntersecting && e.intersectionRatio >= 0.6) setActiveIndex(idx);
   410→      }
   411→    }, opts);
   412→    itemRefs.current.forEach((el) => el && obs.observe(el));
   413→    return () => obs.disconnect();
   414→  }, [items.length]);
   415→
   416→  // Keyboard nav
   417→  useEffect(() => {
   418→    const onKey = (e: KeyboardEvent) => {
   419→      if (e.key === "Escape") onClose();
   420→      else if (e.key === "ArrowUp" || e.key === "ArrowLeft") {
   421→        e.preventDefault();
   422→        scrollToIndex(activeIndex - 1);
   423→      } else if (e.key === "ArrowDown" || e.key === "ArrowRight") {
   424→        e.preventDefault();
   425→        scrollToIndex(activeIndex + 1);
   426→      }
   427→    };
   428→    window.addEventListener("keydown", onKey);
   429→    return () => window.removeEventListener("keydown", onKey);
   430→  }, [activeIndex, onClose]);
   431→
   432→  const scrollToIndex = useCallback((i: number) => {
   433→    const clamped = Math.max(0, Math.min(items.length - 1, i));
   434→    const el = itemRefs.current[clamped];
   435→    if (el) el.scrollIntoView({ behavior: "smooth", block: "start" });
   436→  }, [items.length]);
   437→
   438→  const pokeChrome = () => {
   439→    setShowChrome(true);
   440→    if (hideChromeTimer.current) clearTimeout(hideChromeTimer.current);
   441→    hideChromeTimer.current = setTimeout(() => setShowChrome(false), 3000);
   442→  };
   443→  useEffect(() => {
   444→    pokeChrome();
   445→    return () => { if (hideChromeTimer.current) clearTimeout(hideChromeTimer.current); };
   446→  }, []);
   447→
   448→  return (
   449→    <div
   450→      ref={containerRef}
   451→      className="fixed inset-0 z-50 bg-black overflow-y-auto no-scrollbar reels-feed"
   452→      onMouseMove={pokeChrome}
   453→      onTouchStart={pokeChrome}
   454→    >
   455→      <AnimatePresence>
   456→        {showChrome && (
   457→          <motion.div
   458→            initial={{ opacity: 0, y: -20 }}
   459→            animate={{ opacity: 1, y: 0 }}
   460→            exit={{ opacity: 0, y: -20 }}
   461→            className="fixed top-0 inset-x-0 z-50 p-3 pt-safe bg-gradient-to-b from-black/70 to-transparent flex items-center justify-between pointer-events-none"
   462→          >
   463→            <Button variant="ghost" size="icon" onClick={onClose} className="text-white hover:bg-white/10 pointer-events-auto">
   464→              <X className="w-6 h-6" />
   465→            </Button>
   466→            <div className="text-white text-xs font-medium truncate max-w-[60%] pointer-events-auto">
   467→              {items[activeIndex]?.name} · {activeIndex + 1}/{items.length}
   468→            </div>
   469→            <div className="w-10" />
   470→          </motion.div>
   471→        )}
   472→      </AnimatePresence>
   473→
   474→      <AnimatePresence>
   475→        {showChrome && (
   476→          <>
   477→            <motion.button
   478→              initial={{ opacity: 0, x: 20 }}
   479→              animate={{ opacity: 1, x: 0 }}
   480→              exit={{ opacity: 0, x: 20 }}
   481→              onClick={() => scrollToIndex(activeIndex + 1)}
   482→              disabled={activeIndex >= items.length - 1}
   483→              className="fixed right-4 top-1/2 -translate-y-1/2 z-50 w-10 h-10 rounded-full bg-white/10 hover:bg-white/20 disabled:opacity-30 backdrop-blur flex items-center justify-center text-white"
   484→            >
   485→              <ChevronDown className="w-5 h-5" />
   486→            </motion.button>
   487→            <motion.button
   488→              initial={{ opacity: 0, x: 20 }}
   489→              animate={{ opacity: 1, x: 0 }}
   490→              exit={{ opacity: 0, x: 20 }}
   491→              onClick={() => scrollToIndex(activeIndex - 1)}
   492→              disabled={activeIndex <= 0}
   493→              className="fixed left-4 top-1/2 -translate-y-1/2 z-50 w-10 h-10 rounded-full bg-white/10 hover:bg-white/20 disabled:opacity-30 backdrop-blur flex items-center justify-center text-white"
   494→            >
   495→              <ChevronUp className="w-5 h-5" />
   496→            </motion.button>
   497→          </>
   498→        )}
   499→      </AnimatePresence>
   500→
   501→      {items.map((m, i) => {
   502→        const distance = Math.abs(i - activeIndex);
   503→        const shouldPreload = distance <= 1;
   504→        return (
   505→          <div
   506→            key={m.id}
   507→            data-index={i}
   508→            ref={(el) => { itemRefs.current[i] = el; }}
   509→            className="reels-item w-full flex items-center justify-center bg-black relative"
   510→            style={{ height: "100dvh" }}
   511→          >
   512→            <ReelsVideoItem
   513→              media={m}
   514→              active={i === activeIndex}
   515→              preload={shouldPreload ? "auto" : "metadata"}
   516→            />
   517→          </div>
   518→        );
   519→      })}
   520→    </div>
   521→  );
   522→}

   524→function ReelsVideoItem({
   525→  media,
   526→  active,
   527→  preload,

   598→export function VideoPlayerOverlay({ items, startIndex, reelsMode: initialReelsMode = false, onClose }: Props) {
   599→  const videoPrefs = useUIStore((s) => s.videoPrefs);
   600→  const setVideoPrefs = useUIStore((s) => s.setVideoPrefs);
   601→  const [reelsMode, setReelsMode] = useState(initialReelsMode || videoPrefs?.reelsEnabled || false);
   602→  const [index, setIndex] = useState(startIndex);
   603→  const media = items[index];
   604→
   605→  const toggleReelsMode = () => {
   606→    const newVal = !reelsMode;
   607→    setReelsMode(newVal);
   608→    if (setVideoPrefs && videoPrefs) {
   609→      setVideoPrefs({ ...videoPrefs, reelsEnabled: newVal });
   610→    }
   611→    fetch("/api/user/nav-prefs", {
   612→      method: "PUT",
   613→      headers: { "Content-Type": "application/json" },
   614→      body: JSON.stringify({ reelsEnabled: newVal }),
   615→    }).catch(() => {});
   616→    toast.success(newVal ? "Reels mode enabled" : "Standard mode enabled");
   617→  };
   618→
   619→  if (!media) {
   620→    return (
   621→      <div className="fixed inset-0 bg-black z-50 flex items-center justify-center text-white">
   622→        <p>Video not found</p>
   623→        <Button variant="outline" onClick={onClose} className="ml-3 text-white">Close</Button>
   624→      </div>
   625→    );
   626→  }
   627→
   628→  if (reelsMode) {
   629→    return (
   630→      <div className="fixed inset-0 z-50 bg-black">
   631→        <ReelsFeed items={items} startIndex={startIndex} onClose={onClose} />
   632→        <Button
   633→          variant="ghost"
   634→          size="icon"
   635→          onClick={toggleReelsMode}
   636→          className="fixed top-3 right-3 z-[60] text-white hover:bg-white/10 bg-black/30 backdrop-blur"
   637→          title="Switch to Standard mode"
   638→        >
   639→          <Monitor className="w-5 h-5" />
   640→        </Button>
   641→      </div>
   642→    );
   643→  }
   644→
   645→  return (
   646→    <div className="fixed inset-0 z-50 bg-black">
   647→      <StandardPlayer media={media} onClose={onClose} />
   648→      <Button
   649→        variant="ghost"
   650→        size="icon"
   651→        onClick={toggleReelsMode}
   652→        className="fixed top-3 right-3 z-[60] text-white hover:bg-white/10 bg-black/30 backdrop-blur"
   653→        title="Switch to Reels mode"
   654→      >
   655→        <Smartphone className="w-5 h-5" />
   656→      </Button>
   657→      <div className="fixed bottom-3 left-1/2 -translate-x-1/2 z-[60] flex items-center gap-2 bg-black/50 backdrop-blur px-3 py-1.5 rounded-full">
   658→        <Button
   659→          variant="ghost"
   660→          size="sm"
   661→          disabled={index <= 0}
   662→          onClick={() => setIndex((i) => Math.max(0, i - 1))}
   663→          className="text-white hover:bg-white/10 disabled:opacity-30 h-8 px-2"
   664→        >
   665→          <ChevronUp className="w-4 h-4" />
   666→        </Button>
   667→        <span className="text-white text-xs">{index + 1} / {items.length}</span>
   668→        <Button
   669→          variant="ghost"
   670→          size="sm"
   671→          disabled={index >= items.length - 1}
   672→          onClick={() => setIndex((i) => Math.min(items.length - 1, i + 1))}
   673→          className="text-white hover:bg-white/10 disabled:opacity-30 h-8 px-2"
   674→        >
   675→          <ChevronDown className="w-4 h-4" />
   676→        </Button>
   677→      </div>
   678→    </div>
   679→  );
   680→}
